cmake_minimum_required(VERSION 3.20)
project(CanvasWindowInteraction LANGUAGES CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CANVAS_OUTPUT_DIRECTORY}")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CANVAS_OUTPUT_DIRECTORY}")
endforeach()
add_library(CanvasMouseFilter SHARED MouseFilter.cpp)
target_link_libraries(CanvasMouseFilter PRIVATE user32)
add_executable(CanvasInputHost InputHost.cpp)
target_link_libraries(CanvasInputHost PRIVATE user32)
add_executable(CanvasInputProbe InputProbe.cpp)
target_link_libraries(CanvasInputProbe PRIVATE user32)
foreach(target CanvasMouseFilter CanvasInputHost CanvasInputProbe)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE /W4 /WX)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
